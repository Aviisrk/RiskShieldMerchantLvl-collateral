-- P0a · pricing units (Revenue x100 sanity)
SELECT pricingplan, MIN(pricingvalue) AS min_val, MAX(pricingvalue) AS max_val, COUNT(*) AS n
FROM shield.merchantpricingdetails WHERE product = 'PG' GROUP BY 1;

-- P0b · PG join-key format in shield.txnscreenedresult (composite rows = direct txn join is broken)
--       (v5.6 referenced non-existent s.addeddate / t.txdate; real columns are addedon / txtime, no casts on filter columns)
SELECT COUNT(*) AS pg_rows_30d,
       SUM(CASE WHEN POSITION('_' IN s.transactionid) > 0 THEN 1 ELSE 0 END) AS composite_rows,
       SUM(CASE WHEN t.transactionid IS NOT NULL THEN 1 ELSE 0 END) AS direct_matches
FROM shield.txnscreenedresult s
LEFT JOIN cfsellers.cftransactions t
       ON t.transactionid = s.transactionid
      AND t.txtime >= date_add('day', -35, current_date)
WHERE s.product = 'PG' AND s.addedon >= date_add('day', -30, current_date);

-- P0c · column-name check (silent-zero guard: a wrong column returns 0 rows, not an error)
--       (datashare tables are not listed in information_schema.columns -> use svv_redshift_columns)
SELECT schema_name, table_name, column_name, data_type
FROM svv_redshift_columns
WHERE database_name = 'cashfree_warehouse' AND schema_name IN ('shield', 'cfsellers', 'cfanalytics')
  AND table_name IN ('dotmobiledata', 'riskblacklisteditems', 'txnscreenedresult', 'merchant_rule',
                     'observationscreeningresult', 'ncrpfraudrequests', 'transactionmetadata', 'merchant_metadata')
ORDER BY table_name, column_name;
